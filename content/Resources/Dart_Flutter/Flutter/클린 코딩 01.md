- Flutter로 화면을 구성하는 첫 과제입니다. 화면과 똑같이 구성하는 것도 좋지만 최대한 작은 단위의 Widget 으로 분리해서 개발하는 것을 목표로 해 보아요.
- TODAY 우측 에 적힌 날짜들은 숫자가 반쯤 가려진 것으로 보아 수평으로 스크롤 할 수 있어야 하는 것으로 보입니다.
- 마찬가지로 가장 아래쪽의 `WEEKLY PLANNING`도 아래가 살짝 가려져 있는 것으로 보아 수직으로 스크롤 가능해야 한다는 것을 알 수 있습니다.
- [SingleChildScrollView](https://api.flutter.dev/flutter/widgets/SingleChildScrollView-class.html) 를 사용하면 위젯들을 수평 혹은 수직으로 스크롤 가능한 뷰를 만들 수 있습니다.
- [Collection for](https://dart.dev/language/collections#control-flow-operators) 를 사용하여 반복된 코드를 줄일 수 있습니다.

### 내가 만든 코드

```dart
import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({Key? key}) : super(key: key);

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Flutter Demo',
      theme: ThemeData.dark(), // 대략적인 다크 테마 적용
      home: const HomeScreen(),
      debugShowCheckedModeBanner: false,
    );
  }
}

class HomeScreen extends StatelessWidget {
  const HomeScreen({Key? key}) : super(key: key);

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      // 배경색을 검정색으로 표현
      backgroundColor: Colors.black,
      body: SafeArea(
        child: Padding(
          padding: const EdgeInsets.all(16.0),
          // 세로 방향으로 쌓는 Column
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              // 1) 상단 프로필과 + 버튼, 날짜 표시
              _buildHeader(),
              const SizedBox(height: 20),

              // 2) 일정 카드들을 스크롤할 수 있도록 (예: ListView)
              Expanded(
                child: SingleChildScrollView(
                  child: Column(
                    children: [
                      _buildScheduleCard(
                        title: 'DESIGN MEETING',
                        timeRange: '11:30\n12:20',
                        backgroundColor: Colors.yellow,
                        participants: 'ALEX  HELENA  NANA',
                      ),
                      const SizedBox(height: 16),
                      _buildScheduleCard(
                        title: 'DAILY PROJECT',
                        timeRange: '12:35\n14:10',
                        backgroundColor: Colors.purple,
                        participants: 'ME  RICHARD  CIRY  +4',
                      ),
                      const SizedBox(height: 16),
                      _buildScheduleCard(
                        title: 'WEEKLY PLANNING',
                        timeRange: '15:00\n16:30',
                        backgroundColor: Colors.lightGreen,
                        participants: 'DEN  NANA  MARK',
                      ),
                    ],
                  ),
                ),
              ),
            ],
          ),
        ),
      ),
    );
  }

  // 상단 헤더 부분을 따로 뽑아서 만듦
  Widget _buildHeader() {
    return Row(
      mainAxisAlignment: MainAxisAlignment.spaceBetween,
      children: [
        // 왼쪽: 프로필 사진 + 날짜 표기
        Row(
          children: [
            // 간단한 프로필
            CircleAvatar(
              radius: 20,
              backgroundColor: Colors.grey,
              child: Icon(Icons.person, color: Colors.black),
            ),
            const SizedBox(width: 10),
            // 날짜, 예시로 "MONDAY 16" 표시
            Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: const [
                Text(
                  'MONDAY 16',
                  style: TextStyle(color: Colors.white70, fontSize: 12),
                ),
                // TODAY 와 시간들 표시
                Text(
                  'TODAY · 17  18  19  20',
                  style: TextStyle(color: Colors.white, fontSize: 14),
                ),
              ],
            ),
          ],
        ),
        // 오른쪽: + 버튼
        IconButton(
          onPressed: () {},
          icon: const Icon(Icons.add, color: Colors.white, size: 28),
        ),
      ],
    );
  }

  // 일정 카드 부분을 따로 뽑아서 만드는 메서드
  Widget _buildScheduleCard({
    required String title,
    required String timeRange,
    required Color backgroundColor,
    required String participants,
  }) {
    return Container(
      width: double.infinity,
      // 카드 모양을 주기 위해 조금 둥글게
      decoration: BoxDecoration(
        color: backgroundColor,
        borderRadius: BorderRadius.circular(12),
      ),
      padding: const EdgeInsets.all(16.0),
      child: Row(
        children: [
          // 왼쪽 시간 표시
          SizedBox(
            width: 40,
            child: Text(
              timeRange,
              style: const TextStyle(
                fontSize: 12,
                fontWeight: FontWeight.bold,
                color: Colors.black87,
              ),
              textAlign: TextAlign.right,
            ),
          ),
          const SizedBox(width: 16),
          // 오른쪽 일정 내용
          Expanded(
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                // 일정 제목
                Text(
                  title,
                  style: const TextStyle(
                    fontSize: 24,
                    fontWeight: FontWeight.bold,
                    color: Colors.black,
                  ),
                ),
                const SizedBox(height: 8),
                // 참여자 표시
                Text(
                  participants,
                  style: const TextStyle(
                    fontSize: 14,
                    color: Colors.black87,
                  ),
                ),
              ],
            ),
          ),
        ],
      ),
    );
  }
}

```

```dart
// Header를 만드는 코드 (발췌)
Widget _buildHeader() {
  // 오늘 날짜 구하기
  DateTime now = DateTime.now();
  List<String> weekDays = [
    'MONDAY', 'TUESDAY', 'WEDNESDAY',
    'THURSDAY', 'FRIDAY', 'SATURDAY', 'SUNDAY'
  ];
  String dayString = weekDays[now.weekday - 1]; 
  int dateNumber = now.day; 
  String todayText = '$dayString $dateNumber';  // 예: "MONDAY 16"

  return Row(
    mainAxisAlignment: MainAxisAlignment.spaceBetween,
    children: [
      // 왼쪽: 프로필 사진 + 날짜 표기
      Row(
        children: [
          CircleAvatar(
            radius: 20,
            backgroundColor: Colors.grey,
            child: Icon(Icons.person, color: Colors.black),
          ),
          const SizedBox(width: 10),
          Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              // 1) 위에서 만든 todayText를 표시
              Text(
                todayText,
                style: const TextStyle(color: Colors.white70, fontSize: 12),
              ),
              // 2) TODAY와 시간을 표시하던 부분은 필요에 따라 수정
              const Text(
                'TODAY · 17  18  19  20',
                style: TextStyle(color: Colors.white, fontSize: 14),
              ),
            ],
          ),
        ],
      ),
      IconButton(
        onPressed: () {},
        icon: const Icon(Icons.add, color: Colors.white, size: 28),
      ),
    ],
  );
}
```
